Todos os comandos prontos para copiar e colar direto no Google Cloud Shell (>_ no topo do console do GCP) e na VM de produção. 

export GCP_PROJECT_ID="biteflow-prd"
export GCP_REGION="us-east1"
export GCP_ZONE="us-east1-b"

gcloud projects create $GCP_PROJECT_ID --name="BiteFlow Production"
gcloud config set project $GCP_PROJECT_ID

# Liste sua conta de faturamento e vincule:
gcloud billing accounts list
# gcloud billing projects link $GCP_PROJECT_ID --billing-account=SEU_BILLING_ACCOUNT_ID

gcloud services enable \
  compute.googleapis.com \
  pubsub.googleapis.com \
  artifactregistry.googleapis.com \
        iam.googleapis.com

        gcloud artifacts repositories create biteflow-repo \
        --repository-format=docker \
        --location="$GCP_REGION" \
        --description="BiteFlow Quarkus Native Images" \
        --project="$GCP_PROJECT_ID" || true

No menu Faturamento → Orçamentos e alertas, crie o alerta Alerta Mensal BiteFlow PRD com valor alvo de R$ 30,00 (US$ 5) e notificações em 50%, 80% e 100%. Com e2-micro + 30GB pd-standard em us-east1 e Quarkus Nativo, a VM se enquadra no Always Free Tier do GCP. 

for TOPIC in \
  ifood-order-placed \
  ifood-order-status-changed \
  ifood-chat-message \
  ifood-order-dispute \
  ifood-order-cancellation-requested \
  biteflow-order-status-changed \
  biteflow-order-cancelled \
  biteflow-catalog-changed \
  biteflow-chat-message-outbound; do
  gcloud pubsub topics create "$TOPIC" --project="$GCP_PROJECT_ID" || true
done

gcloud pubsub subscriptions create ifood-order-placed-sub --topic=ifood-order-placed --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create ifood-order-status-changed-sub --topic=ifood-order-status-changed --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create ifood-chat-message-sub --topic=ifood-chat-message --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create biteflow-order-status-changed-sub --topic=biteflow-order-status-changed --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create biteflow-order-cancelled-sub --topic=biteflow-order-cancelled --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create biteflow-catalog-changed-sub --topic=biteflow-catalog-changed --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud pubsub subscriptions create biteflow-chat-message-outbound-sub --topic=biteflow-chat-message-outbound --ack-deadline=60 --project="$GCP_PROJECT_ID"
gcloud iam service-accounts create biteflow-runtime-sa \
  --display-name="BiteFlow Production Runtime SA" \
  --project="$GCP_PROJECT_ID"

SA_EMAIL="biteflow-runtime-sa@${GCP_PROJECT_ID}.iam.gserviceaccount.com"

gcloud projects add-iam-policy-binding "$GCP_PROJECT_ID" --member="serviceAccount:${SA_EMAIL}" --role="roles/pubsub.publisher"
gcloud projects add-iam-policy-binding "$GCP_PROJECT_ID" --member="serviceAccount:${SA_EMAIL}" --role="roles/pubsub.subscriber"
gcloud projects add-iam-policy-binding "$GCP_PROJECT_ID" --member="serviceAccount:${SA_EMAIL}" --role="roles/artifactregistry.reader"
gcloud compute addresses create biteflow-cloud-ip --region="$GCP_REGION" --project="$GCP_PROJECT_ID"
STATIC_IP=$(gcloud compute addresses describe biteflow-cloud-ip --region="$GCP_REGION" --project="$GCP_PROJECT_ID" --format="get(address)")
echo "IP Fixo da API: $STATIC_IP"

gcloud compute firewall-rules create allow-biteflow-https \
  --allow=tcp:80,tcp:443 \
  --target-tags=biteflow-web \
  --project="$GCP_PROJECT_ID"

gcloud compute instances create biteflow-cloud-vm \
  --project="$GCP_PROJECT_ID" \
  --zone="$GCP_ZONE" \
  --machine-type=e2-micro \
  --address="$STATIC_IP" \
  --service-account="$SA_EMAIL" \
  --scopes=https://www.googleapis.com/auth/cloud-platform \
  --tags=biteflow-web \
  --image-family=ubuntu-2404-lts-amd64 \
  --image-project=ubuntu-os-cloud \
  --boot-disk-size=30GB \
  --boot-disk-type=pd-standard
